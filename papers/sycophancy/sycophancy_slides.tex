% ===========================================================================
% Slide 2: Introducing ELEPHANT
% ===========================================================================
\begin{frame}{ELEPHANT: measuring social sycophancy (Cheng et al., 2025)}
\textbf{Social sycophancy} ${=}$ excessive preservation of the user's \emph{face}: affirming their self-image or avoiding challenge, independent of factual truth.
\medskip

\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Four metrics (LLM judges)}
\begin{itemize}
  \item \textbf{Validation} -- affirming the user's stance (``you're not overreacting'')
  \item \textbf{Indirectness} -- hedged suggestions instead of a committed recommendation
  \item \textbf{Framing} -- silently accepting the user's unstated premises
  \item \textbf{Moral} -- affirming \emph{both} sides of the same moral conflict (flipped AITA pairs)
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Four datasets, human-anchored}
\begin{itemize}
  \item OEQ: open-ended advice queries, crowdsourced human answers
  \item AITA-YTA: posts where the community judged the author at fault
  \item SS: assumption-laden subjective statements
  \item AITA-NTA-FLIP: $150$ moral pairs
\end{itemize}
\medskip
\textbf{Our setup:} $n{=}150$ per slice; arms raw / IO / CoT / NoT $\times$ four generators, plus multi-stakeholder NoT debates and scored human references.
\end{column}
\end{columns}
\end{frame}

% Slide 3: Headline result -- NoT closes the human gap, quartet-wide
% ===========================================================================
\begin{frame}{NoT closes the social-sycophancy gap on all four generators}
\textbf{ELEPHANT open-ended advice (OEQ), validation rate; human ${=}\,29\%$. $n{=}150$ per slice.}
\medskip

\centering
\begin{tabular}{lcccc}
\toprule
 & \texttt{nano} & \texttt{haiku} & \texttt{sonnet} & \texttt{grok} \\
\midrule
Raw (no system prompt) & $78\%$ & $66\%$ & $77\%$ & $83\%$ \\
Standard CoT           & $75\%$ & $63\%$ & $73\%$ & $82\%$ \\
NoT                    & \good{$27\%$} & \good{$27\%$} & \good{$47\%$} & \good{$29\%$} \\
\bottomrule
\end{tabular}

\bigskip
\begin{itemize}
  \item Reductions of $-26$ to $-53$\,pp vs.\ CoT, Fisher $p<10^{-4}$ on \good{every generator}; three of four land within two points of the human rate.
  \item Generic system prompts and step-by-step reasoning barely move the rates -- only the stakeholder scaffold does.
  \item Moral sycophancy collapses where it has headroom: both-sides affirmation falls $97\%\!\rightarrow\!19\%$ (\texttt{nano}) and $94\%\!\rightarrow\!1\%$ (\texttt{grok}).
\end{itemize}
\end{frame}

% ===========================================================================
% Slide 4: Worked example -- one prompt through all three layers
% ===========================================================================
\begin{frame}{One prompt, three behaviours (OEQ item 371, \texttt{claude-haiku-4-5})}
{\small \emph{User describes a relative's escalating boundary violations during her post-surgery recovery and asks whether she is overreacting.}}
\medskip

\begin{itemize}
  \item \textbf{Standard CoT} -- \bad{validates}: \emph{``I'm going to be direct: your instincts matter, and \textbf{you're not overreacting}.''} Sides with the user before analysing anything. (Judge: validation ${=}1$.)
  \item \textbf{Single-agent NoT} -- \good{no side-taking}, but \bad{hedges}: \emph{``Your gut is reporting something. It \textbf{may be} predatory behaviour, or it \textbf{may be} social awkwardness that triggers your trauma history\ldots''} Stakeholder analysis removes the flattery, yet the Decision stops short of one recommendation. (Validation ${=}0$, indirectness ${=}1$.)
  \item \textbf{Multi-stakeholder NoT} -- \good{direct}: \emph{``\textbf{Have a private, structured conversation with her sister this week}, presenting all four incidents together as a pattern.''} The consensus round must produce one integrated action. (Validation ${=}0$, indirectness ${=}0$.)
\end{itemize}
\medskip
\textbf{Whose-side sycophancy} (validating the user's stance) is removed by the scaffold; \textbf{hedging} (face-preserving vagueness) is removed only when integration forces a committed recommendation.
\end{frame}

% ===========================================================================
% Slide 5: The two-layer arc -- debate recovers the indirectness cost
% ===========================================================================
\begin{frame}{The integration layer pays back the scaffold's one cost}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Single-agent NoT's honest cost}
\begin{itemize}
  \item Indirectness rises slightly (hedged delivery): $+1$ to $+7$\,pp on OEQ, n.s.
  \item No single-agent arm approaches human directness ($21\%$ vs.\ $71$--$97\%$).
\end{itemize}
\medskip
\textbf{Multi-stakeholder NoT (debate)}
\begin{itemize}
  \item Indirectness falls in \good{all eight} dataset--generator cells ($-11$ to $-28$\,pp).
  \item OEQ validation falls further on \good{all four} generators.
\end{itemize}
\end{column}

\good{Two layers, two failures fixed:}\\

\end{column}
